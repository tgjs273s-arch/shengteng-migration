from pathlib import Path
import hashlib, re, statistics

rows = [(int(s), int(total), float(ms)) for s, total, ms in re.findall(
    r'iteration\s+(\d+)/\s*(\d+).*?elapsed time per iteration \(ms\):\s*([0-9.]+)',
    b.decode('utf-8'))]
window = [ms for step, _, ms in rows if 50 <= step <= 100]
